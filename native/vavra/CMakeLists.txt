# vavra: a Pung-native module (docs/NATIVE-MODULES.md), built from
# dsp/waldorf_engine.cpp with WALDORF_MQ.
# GEARMULATOR_SOURCE: the gearmulator tree's source/ folder (rgeissen/gearmulator).
cmake_minimum_required(VERSION 3.15)
project(vavra CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT GEARMULATOR_SOURCE)
    message(FATAL_ERROR "GEARMULATOR_SOURCE is not set")
endif()
if(NOT SCHWUNG_SRC)
    message(FATAL_ERROR "SCHWUNG_SRC is not set (Schwung's src folder, for host/plugin_api_v1.h)")
endif()
find_package(Threads REQUIRED)
# keep dsp.so unloadable: no STB_GNU_UNIQUE symbols from inline statics
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fno-gnu-unique>)

set(ASMJIT_STATIC TRUE)
set(ASMJIT_NO_INSTALL TRUE)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(${GEARMULATOR_SOURCE}/cpu/dsp56300/source ${CMAKE_BINARY_DIR}/dsp56300)
add_subdirectory(${GEARMULATOR_SOURCE}/framework/baseLib ${CMAKE_BINARY_DIR}/baseLib)
add_subdirectory(${GEARMULATOR_SOURCE}/framework/synthLib ${CMAKE_BINARY_DIR}/synthLib)
add_subdirectory(${GEARMULATOR_SOURCE}/3rdparty/libresample ${CMAKE_BINARY_DIR}/libresample)
add_subdirectory(${GEARMULATOR_SOURCE}/framework/hardwareLib ${CMAKE_BINARY_DIR}/hardwareLib)
add_subdirectory(${GEARMULATOR_SOURCE}/cpu/mc68k ${CMAKE_BINARY_DIR}/mc68k)
target_include_directories(68kEmu PUBLIC ${GEARMULATOR_SOURCE}/cpu)
add_subdirectory(${GEARMULATOR_SOURCE}/waldi/common/wLib ${CMAKE_BINARY_DIR}/wLib)
add_subdirectory(${GEARMULATOR_SOURCE}/waldi/microq/mqLib ${CMAKE_BINARY_DIR}/mqLib)

add_library(vavra-dsp SHARED dsp/waldorf_engine.cpp)
set_target_properties(vavra-dsp PROPERTIES PREFIX "" OUTPUT_NAME "dsp" SUFFIX ".so")
target_compile_definitions(vavra-dsp PRIVATE WALDORF_MQ=1)
target_include_directories(vavra-dsp PRIVATE ${SCHWUNG_SRC} ${SCHWUNG_SRC}/host)
target_link_libraries(vavra-dsp PRIVATE mqLib synthLib Threads::Threads)
target_link_options(vavra-dsp PRIVATE -Wl,--exclude-libs,ALL -static-libstdc++ -static-libgcc)
