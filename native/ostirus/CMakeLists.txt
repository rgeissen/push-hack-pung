# OsTIrus: a Pung-native module (docs/NATIVE-MODULES.md).
# GEARMULATOR_SOURCE: the gearmulator tree's source/ folder (rgeissen/gearmulator).
cmake_minimum_required(VERSION 3.15)
project(ostirus CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT GEARMULATOR_SOURCE)
    message(FATAL_ERROR "GEARMULATOR_SOURCE is not set")
endif()
if(NOT SCHWUNG_SRC)
    message(FATAL_ERROR "SCHWUNG_SRC is not set (Schwung's src folder, for host/plugin_api_v1.h)")
endif()
find_package(Threads REQUIRED)
# keep dsp.so unloadable: no STB_GNU_UNIQUE symbols from inline statics
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fno-gnu-unique>)

set(ASMJIT_STATIC TRUE)
set(ASMJIT_NO_INSTALL TRUE)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(${GEARMULATOR_SOURCE}/cpu/dsp56300/source ${CMAKE_BINARY_DIR}/dsp56300)
add_subdirectory(${GEARMULATOR_SOURCE}/framework/baseLib ${CMAKE_BINARY_DIR}/baseLib)
add_subdirectory(${GEARMULATOR_SOURCE}/framework/synthLib ${CMAKE_BINARY_DIR}/synthLib)
add_subdirectory(${GEARMULATOR_SOURCE}/3rdparty/libresample ${CMAKE_BINARY_DIR}/libresample)
add_subdirectory(${GEARMULATOR_SOURCE}/axel/virusLib ${CMAKE_BINARY_DIR}/virusLib)
target_include_directories(virusLib PUBLIC ${GEARMULATOR_SOURCE}/axel)

add_library(ostirus-dsp SHARED dsp/ostirus_plugin.cpp)
set_target_properties(ostirus-dsp PROPERTIES PREFIX "" OUTPUT_NAME "dsp" SUFFIX ".so")
target_include_directories(ostirus-dsp PRIVATE ${SCHWUNG_SRC} ${SCHWUNG_SRC}/host)
target_link_libraries(ostirus-dsp PRIVATE virusLib synthLib Threads::Threads)
target_link_options(ostirus-dsp PRIVATE -Wl,--exclude-libs,ALL -static-libstdc++ -static-libgcc)

# the emulator, in a process of its own (ostirus_engine.cpp): the plugin starts it
add_executable(ostirus-engine dsp/ostirus_engine.cpp)
target_link_libraries(ostirus-engine PRIVATE virusLib synthLib Threads::Threads)
target_link_options(ostirus-engine PRIVATE -static-libstdc++ -static-libgcc)
